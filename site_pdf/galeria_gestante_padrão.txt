document.addEventListener("DOMContentLoaded", function () {
  const slider = document.getElementById("sliderGestante");

  if (!slider) return;

  const fotos = Array.from(
    slider.querySelectorAll(".slider-gestante-item")
  );

  const botaoAnterior = slider.querySelector(".slider-gestante-anterior");
  const botaoProximo = slider.querySelector(".slider-gestante-proximo");

  let indiceAtual = 0;
  let intervalo;

  function atualizarGaleria() {
    const total = fotos.length;

    fotos.forEach(function (foto, indice) {
      foto.classList.remove("ativo", "anterior", "proximo");

      if (indice === indiceAtual) {
        foto.classList.add("ativo");
      }

      if (indice === (indiceAtual - 1 + total) % total) {
        foto.classList.add("anterior");
      }

      if (indice === (indiceAtual + 1) % total) {
        foto.classList.add("proximo");
      }
    });
  }

  function proximaFoto() {
    indiceAtual = (indiceAtual + 1) % fotos.length;
    atualizarGaleria();
  }

  function fotoAnterior() {
    indiceAtual = (indiceAtual - 1 + fotos.length) % fotos.length;
    atualizarGaleria();
  }

  function iniciarAutoPlay() {
    clearInterval(intervalo);

    intervalo = setInterval(function () {
      proximaFoto();
    }, 3500);
  }

  botaoProximo.addEventListener("click", function () {
    proximaFoto();
    iniciarAutoPlay();
  });

  botaoAnterior.addEventListener("click", function () {
    fotoAnterior();
    iniciarAutoPlay();
  });

  slider.addEventListener("mouseenter", function () {
    clearInterval(intervalo);
  });

  slider.addEventListener("mouseleave", function () {
    iniciarAutoPlay();
  });

  atualizarGaleria();
  iniciarAutoPlay();
});
